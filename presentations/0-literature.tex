                \begin{itemize}

\subsection{Eyeriss}
\begin{frame}{Eyeriss: Row Stationary Dataflow}
    \begin{columns}[c]
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{Row Stationary (RS) Objective}:
                \begin{itemize}
                    \item Minimizes overall data movement energy across all data types simultaneously (ifmaps, weights, psums).
                \end{itemize}
                \vspace{0.5em}
                \item \textbf{1-D Convolution Primitive}:
                \begin{itemize}
                    \item Maps 1 row of weights + 1 row of ifmaps to 1 PE.
                    \item Row pairs stay stationary in local scratchpad (spad).
                \end{itemize}
                \vspace{0.5em}
                \item \textbf{2-D PE Set}:
                \begin{itemize}
                    \item Reuses filter rows horizontally, ifmap rows diagonally.
                    \item Accumulates partial sums (psums) vertically.
                \end{itemize}
            \end{itemize}
        \end{column}

        \begin{column}{0.45\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=\textwidth]{1.2-rowstationary.png}
                \caption{RS Dataflow}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Eyeriss: Architecture}
    \begin{columns}[c]
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{4-Level Memory Hierarchy}:
                \begin{itemize}
                    \item DRAM $\to$ Global Buffer (GLB) $\to$ NoC $\to$ PE spads.
                \end{itemize}
                \vspace{0.5em}
                \item \textbf{Custom NoC Delivery}:
                \begin{itemize}
                    \item Global Input Network for 1-cycle multicast (GLB \textrightarrow PEs).
                    \item Global Output Network for storing psums (PEs \textrightarrow GLB)
                    \item Local Network for vertical psum accumulation.
                \end{itemize}
                \vspace{0.5em}
                \item \textbf{Exploiting sparsity}:
                \begin{itemize}
                    \item \textit{Run-Length Compression (RLC)} saves DRAM bandwidth.
                    \item \textit{Data Gating} check if ifmaps are zero to save power.
                \end{itemize}
            \end{itemize}
        \end{column}

\subsection{TPU}
\begin{frame}{TPU: Architecture \& Systolic Array}
    \begin{columns}[c]
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{Coprocessor Design}:
                \begin{itemize}
                    \item PCIe-attached accelerator; host CPU dispatches CISC instructions (\texttt{MatrixMultiply}, \texttt{Activate} \dots).
                \end{itemize}
                \vspace{0.5em}
                \item \textbf{Matrix Multiply Unit (MMU)}:
                \begin{itemize}
                    \item $256 \times 256$ 8-bit INT MACs.
                    \item 24 MiB Unified Buffer (UB) for activations; 4 MiB Accumulators.
                \end{itemize}
                \vspace{0.5em}
                \item \textbf{Systolic Execution}:
                \begin{itemize}
                    \item Diagonal wavefront dataflow reduces expensive SRAM/UB read-writes.
                \end{itemize}
            \end{itemize}
        \end{column}

        \begin{column}{0.45\textwidth}

\begin{frame}{TPU: Roofline Analysis}
    \begin{columns}[c]
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{Roofline Dynamics}:
                \begin{itemize}
                    \item Ridge point far to the right due to low bandwidth and high peak performance.
                    \item MLPs/LSTMs are memory-bound; CNNs are compute-bound.
                \end{itemize}
                \vspace{0.5em}

\section{VLA Models}
\subsection{Conditional Flow Matching}
\begin{frame}{Preliminaries: Conditional Flow Matching (CFM)}
    \begin{columns}[c]
        \begin{column}{0.6\textwidth}
        \begin{itemize}
            \item $\pi_0$ generates an entire action chunk 
            $\mathbf{A}_t = [a_t,\ldots,a_{t+H-1}]$
            \item \textbf{Training:} a noisy action chunk is constructed as
            $
                \mathbf{A} _t^\tau
                =
                \tau \mathbf{A}_t+(1-\tau)\epsilon,\,
                \epsilon\sim\mathcal{N}(0,I),
            $
            and the network learns the corresponding velocity
            \[
                \mathbf{v}_\theta(\mathbf{A}_t^\tau,\mathbf{o}_t)
                \approx
                \mathbf{u}(\mathbf{A}_t^\tau|\mathbf{A}_t) = 
                \frac{d\mathbf{A}_t^\tau}{d\tau}=
                \mathbf{A}_t-\epsilon,
            \]
            by optimizing the loss
            \[
                \mathcal{L}^\tau(\theta)=
                \mathop{\mathbb{E}}_{\mathbf{A}_t, \mathbf{A}_t^\tau} 
                \Bigl[|| \mathbf{v}_\theta(\mathbf{A}_t^\tau,\mathbf{o}_t) - \mathbf{u}(\mathbf{A}_t^\tau|\mathbf{A}_t) ||^2\Bigr].
            \]
            \item \textbf{Inference:} $\pi_0$ starts from $\mathbf{A}_t^0\sim\mathcal{N}(0,I)$
            and integrates the learned vector field from $\tau=0$ to $\tau=1$:
            \[
                \mathbf{A}_t^{\tau+\delta}
                =
                \mathbf{A}_t^\tau
                +
                \delta\,\mathbf{v}_\theta(\mathbf{A}_t^\tau,\mathbf{o}_t).
            \]
        \end{itemize}
        \end{column}

        \begin{column}{0.45\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=\textwidth]{3.1-CFM.png}
                \caption{CFM visuals.}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}

\subsection{\texorpdfstring{$\pi_0$}{pi\_0}}
\begin{frame}{$\pi_0$: Architecture}
    \begin{columns}[c]
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{Inputs}: $\mathbf{o}_t = [I_t^{1:n} (\text{images}),\, \ell_t (\text{label}),\, \mathbf{q}_t (\text{state})]$
                \item \textbf{Output}: chunk of actions $\mathbf{A}_t = [a_t,\ldots,a_{t+H-1}]$.
                \item \textbf{Architecture}:
                \begin{itemize}
                    \item VLM process images and the instruction.
                    \item Action Expert process robots state and noisy action.
                \end{itemize}
                \vspace{0.4em}
                \item \textbf{Blockwise Causal Attention}:
                \begin{itemize}
                    \item Block 1: $[I_t^{1:n}, \ell_t]$, Block 2: $[q_t]$, Block 3: $[A_t^\tau]$.
                    \item Bidirectional within blocks; causal across blocks.
                    \item Attention between the VLM and the action expert allows for the conditioning on input data.
                \end{itemize}
            \end{itemize}
        \end{column}

        \begin{column}{0.5\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=\textwidth]{3.1-pi0-small.png}
                \caption{$\pi_0$ Architecture}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{$\pi_0$: Training}
    \begin{columns}[c]
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{Efficient KV-Cached Inference}:
                \begin{itemize}
                    \item Compute VLM features + $q_t$ once per observation $o_t$.
                    \item Cache KV tensors for $[I_t^{1:n}, \ell_t, q_t]$.
                    \item 10 Euler steps ($\delta = 0.1$) recompute \textit{only} the 50 action tokens $A_t^\tau$.
                \end{itemize}
                \item \textbf{Cross-embodiment training}: data from different robots is combined into a single model. 
                \item \textbf{Training Recipe:}
                \begin{itemize}
                    \item \textit{Pre-training:} use a very large and diverse dataset.
                    \item \textit{Post-training:} use smaller, high-quality, task-specific datasets.
                \end{itemize}
            \end{itemize}
        \end{column}

\subsection{SmolVLA}
\begin{frame}{SmolVLA: Architecture}
    \begin{columns}[c]
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{VLM}:
                \begin{itemize}
                    \item \textit{Inputs:} input images, instruction, and robot state.
                    \item \textit{Visual tokens:} global image only (no tiling) + pixel shuffle $\to$ get at most 64 tokens/frame.
                    \item \textit{Layer skipping:} Only the first $N=16$ layers are used.

                \item \textbf{Action Expert:} conditional flow matching.
                \item \textbf{Interleaved Attention (CA + SA)}:
                \begin{itemize}
                    \item \textit{Self Attention}: action tokens attend to each other in a causal fashion.
                    \item \textit{Cross Attention}: action tokens attend to the VLM's $K, V$.
                \end{itemize}
            \end{itemize}
        \end{column}

\begin{frame}{SmolVLA: Asynchronous Inference + Data}
    \begin{columns}[c]
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{Decoupled Control Loop}: separates \texttt{RobotClient} execution from \texttt{PolicyServer} inference.
                \item \textbf{Queue Threshold Trigger ($g$)}:
                \begin{itemize}
                    \item Triggers new prediction when remaining queue $|A_t|/n < g$.
                    \item $g=0$: Sequential limit (empty queue causes idle lags).
                    \item $g=0.7$: Async overlap ($\approx 30\%$ faster task completion).
                    \item $g=1.0$: Compute-intensive (process every step).
                \end{itemize}
                \item \textbf{Pretraining Data}
                \begin{itemize}
                    \item community datasets.
                    \item VLM annotation for task descriptions.
                    \item Manual camera viewpoint normalization.

\section{Architectures for VLA}
\subsection{Spec-VLA \& VLA-Cache}
\begin{frame}{Preliminaries: Spec-VLA}
    \begin{columns}[c]
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{Speculative Decoding:} output multiple tokens per step by feeding the main model with a draft answer.
                \item \textbf{Motivation \& Problem}:
                \begin{itemize}
                    \item Autoregressive VLA decoding (OpenVLA) is latency-heavy.
                    \item Exact token matching ($\hat{b} == b$) in standard speculative decoding fails for robotic motor bin tokens.
                \end{itemize}
                \item \textbf{Relaxed Acceptance Mechanism}:
                \begin{itemize}
                    \item Accepts draft bin $\hat{b}$ if within physical bin margin $r$:
                    \[
                    D(a_i, \hat{a}_i) = |b - \hat{b}| \le r
                    \]
                    \item Accepts "synonym" physical motor commands.
                \end{itemize}
            \end{itemize}
        \end{column}

                \begin{itemize}
                    \item Profiling reveals a "stable segment" (steps 20--80) with minimal $L_1$ change.
                    \item Caches prediction output and skips $S-1$ network passes.
                \end{itemize}
                \item \textbf{V-AEFusion (Pipeline Parallelism)}:
                \begin{itemize}
                    \item Overlaps VLM prefill on frame $o_t$ with early denoising steps on Action Expert using stale KV cache from $o_{t-1}$.
                    \item 5 stale steps hide early coarse steps.
                    \item Exploit temporal coherence.
                \end{itemize}
                \item Other optimization techniques: Cache-VLA, Spec-VLA, compiling the model.
            \end{itemize}
        \end{column}
